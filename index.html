let activeUtteranceRef = null; function speakTutorialVoice(rawText, onDone) { if (!('speechSynthesis' in window) || !tutVoiceEnabled) { if (onDone && tutAutoPlay) tutAutoTimer = setTimeout(onDone, 3200); return; } const spoken = rawText .replace(/\bcycle\s*time\b/gi, 'saikel taim') .replace(/\bCT\b/g, 'saikel taim') .replace(/\btakt\s*time\b/gi, 'tak taim') .replace(/\bbottleneck\b/gi, 'botelnek'); const voices = cachedVoices.length ? cachedVoices : window.speechSynthesis.getVoices(); const idVoice = voices.find(v => /id[-_]ID/i.test(v.lang) && /google|natural|neural/i.test(v.name)) || voices.find(v => /id[-_]ID/i.test(v.lang)) || voices.find(v => /^id/i.test(v.lang)); // Hindari cancel() yang memblokir antrean audio di sebagian browser HP if (!window.speechSynthesis.speaking) { window.speechSynthesis.cancel(); } const utt = new SpeechSynthesisUtterance(spoken); activeUtteranceRef = utt; // Cegah Garbage Collection memotong suara di tengah jalan if (idVoice) { utt.voice = idVoice; utt.lang = idVoice.lang; } else { utt.lang = 'id-ID'; } utt.rate = 0.96; utt.pitch = 1.03; utt.volume = 1.0; utt.onend = () => { activeUtteranceRef = null; if (onDone && tutAutoPlay) tutAutoTimer = setTimeout(onDone, 650); }; utt.onerror = () => { activeUtteranceRef = null; if (onDone && tutAutoPlay) tutAutoTimer = setTimeout(onDone, 3000); }; // Panggil langsung di alur klik/render agar tidak terblokir kebijakan audio HP window.speechSynthesis.speak(utt); }
